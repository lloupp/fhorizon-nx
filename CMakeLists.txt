cmake_minimum_required(VERSION 3.20)
project(ForzaHorizonNX LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(fhnx_common INTERFACE)
target_include_directories(fhnx_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/common)
add_executable(fhnx_pc_smoke src/host/main.cpp)
target_link_libraries(fhnx_pc_smoke PRIVATE fhnx_common)
add_executable(fhnx_memory_tests tests/test_guest_memory.cpp)
target_link_libraries(fhnx_memory_tests PRIVATE fhnx_common)
enable_testing()
add_executable(fhnx_graphics_contract tests/test_graphics_contract.cpp)
target_include_directories(fhnx_graphics_contract PRIVATE src/graphics)
add_test(NAME graphics_contract COMMAND fhnx_graphics_contract)
add_test(NAME guest_memory_endianness COMMAND fhnx_memory_tests)
add_test(NAME pc_smoke COMMAND fhnx_pc_smoke)
